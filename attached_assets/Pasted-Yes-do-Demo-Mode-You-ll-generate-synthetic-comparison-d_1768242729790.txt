1) Add this helper (paste into client/src/components/charts/salesDemo.ts)

Create a new file:

client/src/components/charts/salesDemo.ts

export type SalesPoint = {
  label: string;
  dateKey: string;
  total: number;
  food: number;
  bar: number;
  gaming: number;
};

function clamp(n: number, min: number, max: number) {
  return Math.max(min, Math.min(max, n));
}

function rand(seed: number) {
  // deterministic-ish pseudo random (so it doesn’t jump every render)
  let x = Math.sin(seed) * 10000;
  return x - Math.floor(x);
}

/**
 * Build a compare dataset from current dataset:
 * - Keeps same labels/dateKeys so overlay lines up
 * - Applies scale + light noise per point/category
 * - Ensures totals remain >= sum of parts (we recompute total from parts)
 */
export function makeCompareFromCurrent(
  current: SalesPoint[],
  opts?: { scale?: number; volatility?: number; seed?: number }
): SalesPoint[] {
  const scale = opts?.scale ?? 0.92;        // “last period/year was a bit lower”
  const vol = opts?.volatility ?? 0.12;     // +/- noise range
  const seed = opts?.seed ?? 42;

  return (current || []).map((p, i) => {
    const s = seed + i * 13;

    const foodNoise = (rand(s + 1) - 0.5) * 2 * vol;
    const barNoise = (rand(s + 2) - 0.5) * 2 * vol;
    const gamingNoise = (rand(s + 3) - 0.5) * 2 * vol;

    const food = clamp(p.food * scale * (1 + foodNoise), 0, Number.MAX_SAFE_INTEGER);
    const bar = clamp(p.bar * scale * (1 + barNoise), 0, Number.MAX_SAFE_INTEGER);
    const gaming = clamp(p.gaming * scale * (1 + gamingNoise), 0, Number.MAX_SAFE_INTEGER);

    const total = food + bar + gaming;

    return {
      label: p.label,
      dateKey: p.dateKey,
      food: Math.round(food),
      bar: Math.round(bar),
      gaming: Math.round(gaming),
      total: Math.round(total),
    };
  });
}

/**
 * If you have no sales at all yet, generate a demo series.
 * It creates a “normal” restaurant curve with a couple spikes.
 */
export function makeDemoSalesSeries(
  labels: { label: string; dateKey: string }[],
  seed = 7
): SalesPoint[] {
  return labels.map((x, i) => {
    const s = seed + i * 17;

    const base = 600 + 400 * rand(s + 1); // base day
    const spike = (i % 9 === 0 ? 1200 : 0) + (i % 13 === 0 ? 900 : 0);

    const food = Math.round((base + spike) * (0.45 + 0.15 * rand(s + 2)));
    const bar = Math.round((base + spike) * (0.35 + 0.15 * rand(s + 3)));
    const gaming = Math.round((base + spike) * (0.15 + 0.10 * rand(s + 4)));
    const total = food + bar + gaming;

    return { ...x, food, bar, gaming, total };
  });
}

  // Normal behavior
  return {
    current: currentData,
    compare: compareMode !== "off" ? compareData : [],
  };
}, [
  currentSales?.data,
  compareSales?.data,
  demoMode,
  compareMode,
  rangeStart,
  rangeEnd,
  viewBy,

3) If you don’t have buildSalesLabels(...) yet

Add this simple label builder inside Insights (copy/paste) that works for daily/weekly/monthly.

import { addDays, addWeeks, addMonths, format, startOfWeek, startOfMonth } from "date-fns";

function buildSalesLabels(start: Date, end: Date, viewBy: "daily" | "weekly" | "monthly") {
  const labels: { label: string; dateKey: string }[] = [];
  let cursor = new Date(start);

  if (viewBy === "weekly") cursor = startOfWeek(cursor, { weekStartsOn: 1 });
  if (viewBy === "monthly") cursor = startOfMonth(cursor);

  while (cursor <= end) {
    if (viewBy === "daily") {
      labels.push({ label: format(cursor, "MMM d"), dateKey: format(cursor, "yyyy-MM-dd") });
      cursor = addDays(cursor, 1);
    } else if (viewBy === "weekly") {
      labels.push({ label: `Wk ${format(cursor, "II")} ${format(cursor, "yy")}`, dateKey: format(cursor, "yyyy-MM-dd") });
      cursor = addWeeks(cursor, 1);
    } else {
      labels.push({ label: format(cursor, "MMM yyyy"), dateKey: format(cursor, "yyyy-MM-dd") });
      cursor = addMonths(cursor, 1);
    }
  }

  return labels;
}

How this behaves (exactly what you want during build)

Compare Off → no compare series

Compare On but DB has no “last year” → auto-generates compare line (dashed) from current data so the UI still “feels” real

No data at all and Demo Mode ON → generates an entire chart so you can demo the whole Insights page
