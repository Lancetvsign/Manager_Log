// client/src/components/LogAttachmentsSection.tsx
import { useEffect, useState } from "react";

type Attachment = {
  id: number;
  fileName: string;
  fileType: string;
  fileSize: number;
  publicUrl: string;
  createdAt: string;
};

export function LogAttachmentsSection({ logId }: { logId: number }) {
  const [attachments, setAttachments] = useState<Attachment[]>([]);
  const [uploading, setUploading] = useState(false);

  async function load() {
    const res = await fetch(`/api/logs/${logId}/attachments`);
    const data = await res.json();
    setAttachments(data);
  }

  useEffect(() => {
    if (logId) load();
  }, [logId]);

  async function handleFileChange(e: React.ChangeEvent<HTMLInputElement>) {
    const file = e.target.files?.[0];
    if (!file) return;

    const formData = new FormData();
    formData.append("file", file);

    setUploading(true);
    try {
      await fetch(`/api/logs/${logId}/attachments`, {
        method: "POST",
        body: formData,
      });
      await load();
    } finally {
      setUploading(false);
      e.target.value = "";
    }
  }

  async function handleDelete(id: number) {
    if (!window.confirm("Remove this attachment?")) return;
    await fetch(`/api/logs/${logId}/attachments/${id}`, { method: "DELETE" });
    await load();
  }

  return (
    <div className="card mt-4">
      <div className="card-header d-flex justify-content-between align-items-center">
        <h2 className="h6 mb-0">Photos & Attachments</h2>
        <small className="text-muted">Add photos of damage, issues, etc.</small>
      </div>
      <div className="card-body">
        <div className="mb-3">
          <input
            type="file"
            accept="image/*,.pdf,.doc,.docx"
            onChange={handleFileChange}
            disabled={uploading || !logId}
          />
          {!logId && (
            <div className="small text-muted mt-1">
              Save the log entry first, then you can upload attachments.
            </div>
          )}
          {uploading && <div className="small text-muted mt-1">Uploading…</div>}
        </div>

        {attachments.length === 0 ? (
          <p className="small text-muted mb-0">
            No attachments yet. Add photos of broken equipment, damage, etc.
          </p>
        ) : (
          <ul className="list-group">
            {attachments.map((a) => (
              <li
                key={a.id}
                className="list-group-item d-flex justify-content-between align-items-center"
              >
                <div>
                  <div className="fw-semibold">
                    <a href={a.publicUrl} target="_blank" rel="noreferrer">
                      {a.fileName}
                    </a>
                  </div>
                  <div className="small text-muted">
                    {(a.fileSize / 1024).toFixed(1)} KB ·{" "}
                    {new Date(a.createdAt).toLocaleString()}
                  </div>
                </div>
                <button
                  type="button"
                  className="btn btn-link btn-sm text-danger"
                  onClick={() => handleDelete(a.id)}
                >
                  Remove
                </button>
              </li>
            ))}
          </ul>
        )}
      </div>
    </div>
  );
}
