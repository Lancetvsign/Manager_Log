// NotificationSettingsPanel.tsx
import { useEffect, useState } from "react";

interface NotificationList {
  id: number;
  clubId: string;
  name: string;
  description?: string | null;
  isDefault: boolean;
  recipientCount: number;
}

interface NotificationRecipient {
  id: number;
  listId: number;
  name?: string | null;
  email: string;
}

interface NotificationSettingsPanelProps {
  clubId: string;
  onChanged?: () => void; // fired after add/delete so parent can refresh its list
}

export function NotificationSettingsPanel({
  clubId,
  onChanged,
}: NotificationSettingsPanelProps) {
  const [lists, setLists] = useState<NotificationList[]>([]);
  const [selectedListId, setSelectedListId] = useState<number | null>(null);
  const [recipients, setRecipients] = useState<NotificationRecipient[]>([]);
  const [newListName, setNewListName] = useState("");
  const [newRecipientName, setNewRecipientName] = useState("");
  const [newRecipientEmail, setNewRecipientEmail] = useState("");

  async function loadLists() {
    const res = await fetch(`/api/notifications/lists?clubId=${clubId}`);
    const data = await res.json();
    setLists(data);
    if (!selectedListId && data.length) {
      setSelectedListId(data[0].id);
    }
  }

  async function loadRecipients(listId: number) {
    const res = await fetch(`/api/notifications/lists/${listId}/recipients`);
    const data = await res.json();
    setRecipients(data);
  }

  useEffect(() => {
    loadLists();
  }, [clubId]);

  useEffect(() => {
    if (selectedListId) loadRecipients(selectedListId);
    else setRecipients([]);
  }, [selectedListId]);

  async function handleAddList() {
    if (!newListName.trim()) return;
    await fetch("/api/notifications/lists", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ clubId, name: newListName }),
    });
    setNewListName("");
    await loadLists();
    onChanged?.();
  }

  async function handleAddRecipient() {
    if (!selectedListId || !newRecipientEmail.trim()) return;
    await fetch(`/api/notifications/lists/${selectedListId}/recipients`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        name: newRecipientName || undefined,
        email: newRecipientEmail,
      }),
    });
    setNewRecipientName("");
    setNewRecipientEmail("");
    await loadRecipients(selectedListId);
    onChanged?.();
  }

  async function handleDeleteList(id: number) {
    await fetch(`/api/notifications/lists/${id}`, { method: "DELETE" });
    if (selectedListId === id) setSelectedListId(null);
    await loadLists();
    onChanged?.();
  }

  async function handleDeleteRecipient(id: number) {
    await fetch(`/api/notifications/recipients/${id}`, { method: "DELETE" });
    if (selectedListId) {
      await loadRecipients(selectedListId);
    }
    onChanged?.();
  }

  return (
    <div className="grid grid-cols-[1.2fr_2fr] gap-6">
      {/* Lists left */}
      <section>
        <h2 className="text-base font-semibold mb-2">Notification Lists</h2>
        <div className="border rounded-lg overflow-hidden">
          {lists.map((list) => (
            <button
              key={list.id}
              type="button"
              className={
                "w-full px-3 py-2 text-left flex items-center justify-between border-b last:border-b-0" +
                (list.id === selectedListId ? " bg-emerald-50" : "")
              }
              onClick={() => setSelectedListId(list.id)}
            >
              <div>
                <div className="font-medium">
                  {list.name}{" "}
                  {list.isDefault && (
                    <span className="text-xs text-emerald-700 ml-1">
                      Default
                    </span>
                  )}
                </div>
                <div className="text-xs text-muted-foreground">
                  {list.description || "No description"} ·{" "}
                  {list.recipientCount} recipient
                  {list.recipientCount === 1 ? "" : "s"}
                </div>
              </div>
              <button
                type="button"
                className="text-red-500 text-xs"
                onClick={(e) => {
                  e.stopPropagation();
                  handleDeleteList(list.id);
                }}
              >
                🗑
              </button>
            </button>
          ))}
          {lists.length === 0 && (
            <div className="px-3 py-2 text-sm text-muted-foreground">
              No lists yet. Create one below.
            </div>
          )}
        </div>
        <div className="mt-3 flex gap-2">
          <input
            className="flex-1 border rounded px-2 py-1 text-sm"
            placeholder="New list name (e.g. Core Team)"
            value={newListName}
            onChange={(e) => setNewListName(e.target.value)}
          />
          <button
            type="button"
            className="px-3 py-1 text-sm rounded bg-emerald-600 text-white"
            onClick={handleAddList}
          >
            Add
          </button>
        </div>
      </section>

      {/* Recipients right */}
      <section>
        <h2 className="text-base font-semibold mb-2">
          Recipients{selectedListId
            ? ` for "${lists.find((l) => l.id === selectedListId)?.name ?? ""}"`
            : ""}
        </h2>

        {selectedListId ? (
          <>
            <table className="w-full text-sm border rounded overflow-hidden">
              <thead className="bg-slate-50 text-left">
                <tr>
                  <th className="px-3 py-2">Name</th>
                  <th className="px-3 py-2">Email</th>
                  <th className="px-3 py-2"></th>
                </tr>
              </thead>
              <tbody>
                {recipients.map((r) => (
                  <tr key={r.id} className="border-t">
                    <td className="px-3 py-1">{r.name || "—"}</td>
                    <td className="px-3 py-1">{r.email}</td>
                    <td className="px-3 py-1 text-right">
                      <button
                        type="button"
                        className="text-xs text-red-500"
                        onClick={() => handleDeleteRecipient(r.id)}
                      >
                        Remove
                      </button>
                    </td>
                  </tr>
                ))}
                {recipients.length === 0 && (
                  <tr>
                    <td
                      colSpan={3}
                      className="px-3 py-2 text-sm text-muted-foreground"
                    >
                      No recipients yet. Add one below.
                    </td>
                  </tr>
                )}
              </tbody>
            </table>

            <div className="mt-3 flex gap-2">
              <input
                className="flex-1 border rounded px-2 py-1 text-sm"
                placeholder="Name (optional)"
                value={newRecipientName}
                onChange={(e) => setNewRecipientName(e.target.value)}
              />
              <input
                className="flex-1 border rounded px-2 py-1 text-sm"
                placeholder="Email"
                value={newRecipientEmail}
                onChange={(e) => setNewRecipientEmail(e.target.value)}
              />
              <button
                type="button"
                className="px-3 py-1 text-sm rounded bg-slate-900 text-white"
                onClick={handleAddRecipient}
              >
                Add
              </button>
            </div>
          </>
        ) : (
          <p className="text-sm text-muted-foreground">
            Select a list on the left to manage its recipients.
          </p>
        )}
      </section>
    </div>
  );
}
