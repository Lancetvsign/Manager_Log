Below is a drop-in approach you can paste into Replit (SQLite + API + React/TS). Adjust names to match your project (models/routes folder names, etc.).

1) Data model (SQLite)

Create a table to store default email content settings per club.

-- db migration
CREATE TABLE IF NOT EXISTS email_settings (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  club_id TEXT NOT NULL UNIQUE,

  -- content toggles
  include_header INTEGER NOT NULL DEFAULT 1,          -- date/day/managers
  include_weather INTEGER NOT NULL DEFAULT 1,
  include_sales INTEGER NOT NULL DEFAULT 1,
  include_drawing INTEGER NOT NULL DEFAULT 1,
  include_operational_notes INTEGER NOT NULL DEFAULT 1,
  include_maintenance_notes INTEGER NOT NULL DEFAULT 1,
  include_maintenance_list INTEGER NOT NULL DEFAULT 1,

  -- optional: attachments/links
  include_photos INTEGER NOT NULL DEFAULT 1,

  -- optional: per-list defaults (JSON string)
  list_defaults_json TEXT NOT NULL DEFAULT '{}',

  updated_at TEXT NOT NULL DEFAULT (datetime('now'))
);


list_defaults_json lets you store “Core Team defaults to Full Log” and “Kitchen Staff defaults to Ops Notes only”, etc.

Example value:

{
  "Core Team": "full",
  "Kitchen Staff": "ops_only",
  "Ownership": "sales_only"
}

2) Server API: GET/PUT email settings

Create a route like server/routes/emailSettings.ts (or wherever your routes live).

import express from "express";
import { db } from "../db"; // adjust import to your sqlite helper

const router = express.Router();

router.get("/api/email-settings", (req, res) => {
  const clubId = String(req.query.clubId || "");
  if (!clubId) return res.status(400).json({ error: "clubId required" });

  const row = db.prepare(`SELECT * FROM email_settings WHERE club_id = ?`).get(clubId);

  if (!row) {
    // return defaults if not yet created
    return res.json({
      clubId,
      includeHeader: true,
      includeWeather: true,
      includeSales: true,
      includeDrawing: true,
      includeOperationalNotes: true,
      includeMaintenanceNotes: true,
      includeMaintenanceList: true,
      includePhotos: true,
      listDefaults: {},
    });
  }

  res.json({
    clubId,
    includeHeader: !!row.include_header,
    includeWeather: !!row.include_weather,
    includeSales: !!row.include_sales,
    includeDrawing: !!row.include_drawing,
    includeOperationalNotes: !!row.include_operational_notes,
    includeMaintenanceNotes: !!row.include_maintenance_notes,
    includeMaintenanceList: !!row.include_maintenance_list,
    includePhotos: !!row.include_photos,
    listDefaults: safeJson(row.list_defaults_json, {}),
  });
});

router.put("/api/email-settings", express.json(), (req, res) => {
  const {
    clubId,
    includeHeader,
    includeWeather,
    includeSales,
    includeDrawing,
    includeOperationalNotes,
    includeMaintenanceNotes,
    includeMaintenanceList,
    includePhotos,
    listDefaults,
  } = req.body || {};

  if (!clubId) return res.status(400).json({ error: "clubId required" });

  const stmt = db.prepare(`
    INSERT INTO email_settings (
      club_id,
      include_header, include_weather, include_sales, include_drawing,
      include_operational_notes, include_maintenance_notes, include_maintenance_list,
      include_photos, list_defaults_json, updated_at
    ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, datetime('now'))
    ON CONFLICT(club_id) DO UPDATE SET
      include_header = excluded.include_header,
      include_weather = excluded.include_weather,
      include_sales = excluded.include_sales,
      include_drawing = excluded.include_drawing,
      include_operational_notes = excluded.include_operational_notes,
      include_maintenance_notes = excluded.include_maintenance_notes,
      include_maintenance_list = excluded.include_maintenance_list,
      include_photos = excluded.include_photos,
      list_defaults_json = excluded.list_defaults_json,
      updated_at = datetime('now')
  `);

  stmt.run(
    clubId,
    boolToInt(includeHeader),
    boolToInt(includeWeather),
    boolToInt(includeSales),
    boolToInt(includeDrawing),
    boolToInt(includeOperationalNotes),
    boolToInt(includeMaintenanceNotes),
    boolToInt(includeMaintenanceList),
    boolToInt(includePhotos),
    JSON.stringify(listDefaults || {})
  );

  res.json({ ok: true });
});

function boolToInt(v: any) {
  return v ? 1 : 0;
}

function safeJson(str: string, fallback: any) {
  try { return JSON.parse(str); } catch { return fallback; }
}

export default router;

3) Email builder: generate message based on toggles

Wherever you currently build the email “Message” field, swap it to use a function like this.

type EmailContentToggles = {
  includeHeader: boolean;
  includeWeather: boolean;
  includeSales: boolean;
  includeDrawing: boolean;
  includeOperationalNotes: boolean;
  includeMaintenanceNotes: boolean;
  includeMaintenanceList: boolean;
  includePhotos: boolean;
};

export function buildLogEmailBody(params: {
  log: any;                  // your log row
  maintenanceItems: any[];   // running maintenance list
  photoLinks?: { label: string; url: string }[];
  toggles: EmailContentToggles;
  logUrl?: string;
}) {
  const { log, maintenanceItems, photoLinks = [], toggles, logUrl } = params;
  const lines: string[] = [];

  if (toggles.includeHeader) {
    lines.push(`Date: ${log.date}`);
    lines.push(`Manager(s): ${log.manager_name || "Not specified"}`);
    if (log.am_manager_name || log.pm_manager_name || log.kitchen_manager_name) {
      lines.push(
        `AM: ${log.am_manager_name || "-"} | PM: ${log.pm_manager_name || "-"} | Kitchen: ${log.kitchen_manager_name || "-"}`
      );
    }
    lines.push("");
  }

  if (toggles.includeWeather) {
    lines.push("Weather:");
    lines.push(`- Today: ${log.weather_today_summary || "Unavailable"}`);
    lines.push(`- Last Year: ${log.weather_last_year_summary || "Unavailable"}`);
    lines.push("");
  }

  if (toggles.includeSales) {
    lines.push("Sales:");
    lines.push(`- Food: $${num(log.food_sales)}`);
    lines.push(`- Bar: $${num(log.bar_sales)}`);
    lines.push(`- Gaming: $${num(log.gaming_sales)}`);
    lines.push(`- Total: $${num(log.total_sales)}`);
    lines.push("");
  }

  if (toggles.includeDrawing) {
    lines.push("Drawing:");
    lines.push(`- Daily: ${log.daily_number ?? "-"}`);
    lines.push(`- Weekly: ${log.weekly_number ?? "-"}`);
    lines.push("");
  }

  if (toggles.includeOperationalNotes) {
    lines.push("Operational Notes:");
    lines.push(log.operational_notes?.trim() ? log.operational_notes.trim() : "-");
    lines.push("");
  }

  if (toggles.includeMaintenanceNotes) {
    lines.push("Maintenance Notes:");
    lines.push(log.maintenance_notes?.trim() ? log.maintenance_notes.trim() : "-");
    lines.push("");
  }

  if (toggles.includeMaintenanceList) {
    lines.push("Maintenance List:");
    const openOrRecent = maintenanceItems?.length ? maintenanceItems : [];
    if (openOrRecent.length === 0) {
      lines.push("-");
    } else {
      for (const item of openOrRecent) {
        // include added date + status (you can format nicer later)
        lines.push(`- [${item.status}] ${item.description} (added ${item.created_at || item.date_added || ""})`);
      }
    }
    lines.push("");
  }

  if (toggles.includePhotos) {
    lines.push("Photos:");
    if (!photoLinks.length) lines.push("-");
    else photoLinks.forEach(p => lines.push(`- ${p.label}: ${p.url}`));
    lines.push("");
  }

  if (logUrl) {
    lines.push(`Log link: ${logUrl}`);
  }

  return lines.join("\n");
}

function num(v: any) {
  const n = Number(v || 0);
  return n.toFixed(2);
}

4) Frontend: add “Email Content” toggles inside the Email modal Settings

You already have a “Settings” accordion. Inside it, add:

Presets (Full log / Ops only / Maintenance only / Sales only)

Checkboxes (includeWeather/includeSales/etc.)

Save button (persists to /api/email-settings)

State + load/save
type EmailSettings = {
  includeHeader: boolean;
  includeWeather: boolean;
  includeSales: boolean;
  includeDrawing: boolean;
  includeOperationalNotes: boolean;
  includeMaintenanceNotes: boolean;
  includeMaintenanceList: boolean;
  includePhotos: boolean;
  listDefaults: Record<string, string>;
};

const DEFAULT_SETTINGS: EmailSettings = {
  includeHeader: true,
  includeWeather: true,
  includeSales: true,
  includeDrawing: true,
  includeOperationalNotes: true,
  includeMaintenanceNotes: true,
  includeMaintenanceList: true,
  includePhotos: true,
  listDefaults: {},
};

const PRESETS: Record<string, Partial<EmailSettings>> = {
  full: DEFAULT_SETTINGS,
  ops_only: {
    includeHeader: true,
    includeOperationalNotes: true,
    includeWeather: false,
    includeSales: false,
    includeDrawing: false,
    includeMaintenanceNotes: false,
    includeMaintenanceList: false,
    includePhotos: false,
  },
  maintenance_only: {
    includeHeader: true,
    includeMaintenanceNotes: true,
    includeMaintenanceList: true,
    includeOperationalNotes: false,
    includeWeather: false,
    includeSales: false,
    includeDrawing: false,
  },
  sales_only: {
    includeHeader: true,
    includeSales: true,
    includeOperationalNotes: false,
    includeMaintenanceNotes: false,
    includeMaintenanceList: false,
    includeWeather: false,
    includeDrawing: false,
    includePhotos: false,
  },
};

function applyPreset(current: EmailSettings, presetKey: string): EmailSettings {
  const patch = PRESETS[presetKey] || {};
  return { ...current, ...patch };
}

const [emailSettings, setEmailSettings] = useState<EmailSettings>(DEFAULT_SETTINGS);
const [loadingEmailSettings, setLoadingEmailSettings] = useState(false);
const [savingEmailSettings, setSavingEmailSettings] = useState(false);

useEffect(() => {
  if (!clubId) return;
  (async () => {
    setLoadingEmailSettings(true);
    const res = await fetch(`/api/email-settings?clubId=${clubId}`);
    const data = await res.json();
    setEmailSettings({ ...DEFAULT_SETTINGS, ...data });
    setLoadingEmailSettings(false);
  })();
}, [clubId]);

async function saveEmailSettings() {
  setSavingEmailSettings(true);
  await fetch(`/api/email-settings`, {
    method: "PUT",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ clubId, ...emailSettings }),
  });
  setSavingEmailSettings(false);
}
