// shared/schema.ts

export const maintenanceSnapshots = pgTable(
  "maintenance_snapshots",
  {
    id: serial("id").primaryKey(),
    clubId: varchar("club_id", { length: 64 }).notNull(),
    logId: integer("log_id").notNull(), // points to the day’s log

    // which maintenance item was open that day
    todoItemId: integer("todo_item_id").notNull(),

    // denormalized fields so history stays accurate even if item text changes later
    description: text("description").notNull(),
    status: varchar("status", { length: 20 }).notNull(), // open/completed at snapshot time
    notes: text("notes"),

    // optional: how many days it has rolled at the time of snapshot
    daysOpen: integer("days_open").notNull().default(0),

    createdAt: timestamp("created_at").notNull().defaultNow(),
  },
  (t) => ({
    clubLogIdx: index("maintenance_snapshots_club_log_idx").on(t.clubId, t.logId),
    clubTodoIdx: index("maintenance_snapshots_club_todo_idx").on(t.clubId, t.todoItemId),
  })
);

B) Server: create snapshots when a log is saved

In server/routes.ts, inside the “save/update log” handler (where the log entry is created/updated), add this after the log is saved:

// server/routes.ts
import { and, eq, gte, lte, sql, desc, count } from "drizzle-orm";
import { db } from "./db";
import { todoItems, maintenanceSnapshots, managerLogs } from "@shared/schema";

// helper: create a daily snapshot for all OPEN items
async function snapshotOpenMaintenanceForLog(clubId: string, logId: number) {
  // 1) clear any existing snapshots for that log (idempotent)
  await db.delete(maintenanceSnapshots).where(and(
    eq(maintenanceSnapshots.clubId, clubId),
    eq(maintenanceSnapshots.logId, logId)
  ));

  // 2) load all open maintenance items (oldest → newest)
  const openItems = await db
    .select()
    .from(todoItems)
    .where(and(eq(todoItems.clubId, clubId), eq(todoItems.status, "open")))
    .orderBy(todoItems.createdAt);

  // 3) insert snapshots
  if (openItems.length > 0) {
    await db.insert(maintenanceSnapshots).values(
      openItems.map((it) => ({
        clubId,
        logId,
        todoItemId: it.id,
        description: it.description,
        status: it.status,
        notes: it.notes ?? null,
        daysOpen: 0, // (optional) you can compute below
      }))
    );
  }
}

C) Server: Maintenance Insights endpoint

Add this endpoint to server/routes.ts:

// server/routes.ts

app.get("/api/insights/maintenance", async (req, res) => {
  try {
    const clubId = String(req.query.clubId || "");
    const start = String(req.query.start || "");
    const end = String(req.query.end || "");

    if (!clubId || !start || !end) {
      return res.status(400).json({ error: "clubId, start, end are required" });
    }

    const startDate = new Date(start);
    const endDate = new Date(end);

    // 1) OPEN NOW (live)
    const openNowRow = await db
      .select({ c: count() })
      .from(todoItems)
      .where(and(eq(todoItems.clubId, clubId), eq(todoItems.status, "open")));
    const openNow = Number(openNowRow?.[0]?.c ?? 0);

    // 2) COMPLETED IN RANGE (live, based on completedAt)
    const completedRow = await db
      .select({ c: count() })
      .from(todoItems)
      .where(
        and(
          eq(todoItems.clubId, clubId),
          sql`${todoItems.completedAt} is not null`,
          gte(todoItems.completedAt, startDate),
          lte(todoItems.completedAt, endDate)
        )
      );
    const completedInRange = Number(completedRow?.[0]?.c ?? 0);

    // 3) AVG DAYS OPEN (for currently open items)
    const avgDaysOpenRow = await db
      .select({
        avgDays: sql<number>`avg(extract(epoch from (now() - ${todoItems.createdAt})) / 86400)`,
      })
      .from(todoItems)
      .where(and(eq(todoItems.clubId, clubId), eq(todoItems.status, "open")));
    const avgDaysOpen = Number(avgDaysOpenRow?.[0]?.avgDays ?? 0);

    // 4) DAILY OPEN COUNTS (snapshots grouped by log date)
    const dailyOpenCounts = await db
      .select({
        date: managerLogs.date, // assumes managerLogs has a date column
        count: count(),
      })
      .from(maintenanceSnapshots)
      .innerJoin(managerLogs, eq(managerLogs.id, maintenanceSnapshots.logId))
      .where(
        and(
          eq(maintenanceSnapshots.clubId, clubId),
          gte(managerLogs.date, startDate),
          lte(managerLogs.date, endDate)
        )
      )
      .groupBy(managerLogs.date)
      .orderBy(managerLogs.date);

    // 5) DAILY COMPLETIONS (by completedAt date)
    const dailyCompletions = await db
      .select({
        date: sql<string>`date(${todoItems.completedAt})`,
        count: count(),
      })
      .from(todoItems)
      .where(
        and(
          eq(todoItems.clubId, clubId),
          sql`${todoItems.completedAt} is not null`,
          gte(todoItems.completedAt, startDate),
          lte(todoItems.completedAt, endDate)
        )
      )
      .groupBy(sql`date(${todoItems.completedAt})`)
      .orderBy(sql`date(${todoItems.completedAt})`);

    // 6) TOP OLDEST OPEN
    const oldestOpen = await db
      .select({
        id: todoItems.id,
        description: todoItems.description,
        createdAt: todoItems.createdAt,
      })
      .from(todoItems)
      .where(and(eq(todoItems.clubId, clubId), eq(todoItems.status, "open")))
      .orderBy(todoItems.createdAt)
      .limit(5);

    const topOldestOpen = oldestOpen.map((it) => ({
      id: it.id,
      description: it.description,
      daysOpen: Math.floor((Date.now() - new Date(it.createdAt).getTime()) / 86400000),
    }));

    // 7) REPEATED CARRYOVERS (items that appear in snapshots multiple days in range)
    const repeated = await db
      .select({
        todoItemId: maintenanceSnapshots.todoItemId,
        description: sql<string>`max(${maintenanceSnapshots.description})`,
        daysAppeared: count(),
      })
      .from(maintenanceSnapshots)
      .innerJoin(managerLogs, eq(managerLogs.id, maintenanceSnapshots.logId))
      .where(
        and(
          eq(maintenanceSnapshots.clubId, clubId),
          gte(managerLogs.date, startDate),
          lte(managerLogs.date, endDate)
        )
      )
      .groupBy(maintenanceSnapshots.todoItemId)
      .having(sql`count(*) > 1`)
      .orderBy(sql`count(*) desc`)
      .limit(10);

    return res.json({
      openNow,
      completedInRange,
      avgDaysOpen: Number.isFinite(avgDaysOpen) ? avgDaysOpen : 0,
      dailyOpenCounts,
      dailyCompletions,
      topOldestOpen,
      repeatedCarryovers: repeated.map((r) => ({
        id: r.todoItemId,
        description: r.description,
        daysAppeared: Number(r.daysAppeared),
      })),
    });
  } catch (e) {
    console.error(e);
    res.status(500).json({ error: "Failed to load maintenance insights" });
  }
});
