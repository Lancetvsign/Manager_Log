4) Frontend (React + Tailwind)
A) Completed label helper
function completedLabel(completedAt) {
  if (!completedAt) return "";
  const d = new Date(completedAt);
  const now = new Date();
  const sameDay = d.toDateString() === now.toDateString();
  return sameDay
    ? "Completed Today"
    : `Completed ${d.toLocaleDateString()} ${d.toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" })}`;
}

B) Button handlers (with confirmation)
async function completeItem(item) {
  const ok = window.confirm(`Mark "${item.description}" as completed?`);
  if (!ok) return;

  await fetch(`/api/maintenance/${item.id}/complete`, { method: "POST" });
  await refreshMaintenance(); // whatever your reload function is
}

async function reopenItem(item) {
  const ok = window.confirm(`Undo completion for "${item.description}"?`);
  if (!ok) return;

  await fetch(`/api/maintenance/${item.id}/reopen`, { method: "POST" });
  await refreshMaintenance();
}

5) Sorting rule (your spec: oldest → newest, open items)

If you want Open first then Completed at bottom, still oldest → newest inside each group:

SELECT *
FROM maintenance_items
ORDER BY
  CASE WHEN status='open' THEN 0 ELSE 1 END,
  date_added ASC;


(Replace date_added with your actual created date column.)
