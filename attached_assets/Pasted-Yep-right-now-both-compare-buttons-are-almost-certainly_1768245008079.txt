This Week vs This Week Last Year → shift 364 days back (52 weeks) so Friday stays Friday (NOT 365)

1) Copy/paste helpers (put near the top of SalesChart.tsx)
type CompareMode = "off" | "lastWeek" | "lastYearSameWeekday";

const shiftDateByDays = (d: Date, days: number) => {
  const x = new Date(d);
  x.setDate(x.getDate() + days);
  return x;
};

// 52-week shift keeps weekday alignment (Fri -> Fri)
const getCompareShiftDays = (mode: CompareMode) => {
  if (mode === "lastWeek") return -7;
  if (mode === "lastYearSameWeekday") return -364;
  return 0;
};

// deterministic "random" (so demo is stable)
const mulberry32 = (seed: number) => {
  return () => {
    let t = (seed += 0x6d2b79f5);
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
};

// Builds demo sales series for N days starting at startDate.
// Values are intentionally shaped + noisy so comparisons are obvious.
const buildDemoSalesSeries = (startDate: Date, days: number, seed: number) => {
  const rnd = mulberry32(seed);

  const points = [];
  for (let i = 0; i < days; i++) {
    const date = shiftDateByDays(startDate, i);

    // weekly seasonality + a small trend + noise
    const weekday = date.getDay(); // 0..6
    const weekendBoost = weekday === 5 || weekday === 6 ? 1.25 : weekday === 0 ? 0.9 : 1.0;

    const trend = 1 + i * 0.01; // gentle climb
    const noise = 0.85 + rnd() * 0.35; // 0.85..1.20

    const base = 1200 * weekendBoost * trend * noise;

    // split into categories with slightly different wobble
    const food = Math.round(base * (0.45 + rnd() * 0.10));
    const bar = Math.round(base * (0.35 + rnd() * 0.10));
    const gaming = Math.round(base * (0.15 + rnd() * 0.10));

    const total = food + bar + gaming;

    points.push({
      date, // keep raw Date for any formatting you do elsewhere
      food,
      bar,
      gaming,
      total,
    });
  }

  return points;
};
