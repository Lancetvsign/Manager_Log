2) Create/Replace src/components/DailyManagerLogHeader.jsx
import React, { useMemo } from "react";
import WeatherWide from "./WeatherWide";

function formatLongDate(dateStr) {
  if (!dateStr) return "";
  const d = new Date(dateStr + "T00:00:00");
  return d.toLocaleDateString(undefined, {
    year: "numeric",
    month: "long",
    day: "numeric",
  });
}

function formatWeekday(dateStr) {
  if (!dateStr) return "";
  const d = new Date(dateStr + "T00:00:00");
  return d.toLocaleDateString(undefined, { weekday: "long" });
}

export default function DailyManagerLogHeader({
  // required
  date,
  setDate,

  amManager,
  setAmManager,
  pmManager,
  setPmManager,
  kitchenManager,
  setKitchenManager,

  today,

  const weekday = useMemo(() => formatWeekday(date), [date]);

  return (

      {/* TOP: left content + right status (daily/weekly + weather) */}
      <div className="grid grid-cols-1 gap-6 lg:grid-cols-[1fr_440px]">
        {/* LEFT */}
        <div>
          <h2 className="text-lg font-semibold text-white">
            Daily Manager Log Entry
          </h2>

          {/* Date + computed day */}

              Date
            </label>

            <div className="flex flex-wrap items-center gap-3">
              <input
                type="date"
                value={date || ""}
                onChange={(e) => setDate?.(e.target.value)}
                className="h-9 w-[190px] rounded-lg border border-white/[0.08] bg-black/20 px-3 text-sm text-white outline-none focus:border-white/25"
              />

              <div className="flex flex-col leading-tight">
                <div className="text-sm text-white/90">
                  {longDate || "Select a date"}
                </div>
                <div className="text-xs text-white/60">{weekday || ""}</div>
              </div>
            </div>
          </div>
        </div>

        {/* RIGHT */}
        <div className="flex flex-col items-end gap-3">
          {/* Daily/Weekly moved to top-right */}
          <div className="flex items-center gap-3">
            <div className="rounded-xl border border-white/[0.08] bg-black/15 px-3 py-2">
              <div className="text-[10px] uppercase tracking-wide text-white/50">
                Daily #
              </div>
              <div className="text-sm font-semibold text-white">
                {dailyNumber ?? "—"}
              </div>
            </div>

            <div className="rounded-xl border border-white/[0.08] bg-black/15 px-3 py-2">

                Weekly #
              </div>

              hi={lastYear?.hi ?? "—"}

              icon={lastYear?.icon ?? "🕰️"}
              muted

function ManagerRow({ label, value, onChange }) {
  return (
    <div className="flex items-center gap-2 rounded-xl border border-white/[0.08] bg-black/10 px-3 py-2.5">
      {/* reduced spacing between label and input */}
      <div className="w-10 text-[11px] font-semibold uppercase tracking-wide text-white/60">
        {label}
      </div>

      <input
        value={value || ""}
        onChange={(e) => onChange?.(e.target.value)}
        placeholder="Name"
        className="h-9 w-full rounded-md border border-white/[0.08] bg-black/20 px-3 text-sm text-white outline-none focus:border-white/25"
      />
    </div>
  );
}
