✅ React state (series toggles)

Drop this into the Sales chart component:

const [visible, setVisible] = useState({
  total: true,
  food: true,
  bar: true,
  gaming: true,
});

function toggleSeries(key: keyof typeof visible) {
  setVisible((v) => ({ ...v, [key]: !v[key] }));
}

UI chips (Tailwind)
const chipBase =
  "px-3 py-1 rounded-md text-xs font-medium border transition";
const onChip =
  "bg-emerald-500/15 border-emerald-400/40 text-emerald-200";
const offChip =
  "bg-white/5 border-white/10 text-white/60 hover:bg-white/10";

<div className="flex flex-wrap gap-2">
  {(["total","food","bar","gaming"] as const).map((k) => (
    <button
      key={k}
      type="button"
      onClick={() => toggleSeries(k)}
      className={`${chipBase} ${visible[k] ? onChip : offChip}`}
    >
      {k.toUpperCase()}
    </button>
  ))}
</div>

OPTION A — Recharts implementation (recommended)
Main chart that respects toggles
import {
  ResponsiveContainer,
  LineChart, Line,
  AreaChart, Area,
  BarChart, Bar,
  XAxis, YAxis, Tooltip, CartesianGrid,
} from "recharts";

function SalesChart({ data, chartType }: { data: SalesPoint[]; chartType: "line"|"bar"|"area" }) {
  const series = [
    { key: "total", label: "Total" },
    { key: "food", label: "Food" },
    { key: "bar", label: "Bar" },
    { key: "gaming", label: "Gaming" },
  ] as const;

  const common = (
    <>
      <CartesianGrid strokeDasharray="3 3" opacity={0.15} />
      <XAxis dataKey="date" tick={{ fill: "rgba(255,255,255,0.6)", fontSize: 12 }} />
      <YAxis tick={{ fill: "rgba(255,255,255,0.6)", fontSize: 12 }} />
      <Tooltip />
    </>
  );

  return (
    <div className="h-72 w-full">
      <ResponsiveContainer>
        {chartType === "line" ? (
          <LineChart data={data}>
            {common}
            {visible.total && <Line dataKey="total" dot={false} />}
            {visible.food && <Line dataKey="food" dot={false} />}
            {visible.bar && <Line dataKey="bar" dot={false} />}
            {visible.gaming && <Line dataKey="gaming" dot={false} />}
          </LineChart>
        ) : chartType === "area" ? (
          <AreaChart data={data}>
            {common}
            {visible.total && <Area dataKey="total" fillOpacity={0.15} />}
            {visible.food && <Area dataKey="food" fillOpacity={0.15} />}
            {visible.bar && <Area dataKey="bar" fillOpacity={0.15} />}
            {visible.gaming && <Area dataKey="gaming" fillOpacity={0.15} />}
          </AreaChart>
        ) : (
          <BarChart data={data}>
            {common}
            {visible.total && <Bar dataKey="total" />}
            {visible.food && <Bar dataKey="food" />}
            {visible.bar && <Bar dataKey="bar" />}
            {visible.gaming && <Bar dataKey="gaming" />}
          </BarChart>
        )}
      </ResponsiveContainer>
    </div>
  );
}
