import os
import sqlite3
from datetime import datetime
from flask import Flask, request, redirect, url_for, render_template_string, abort

app = Flask(__name__)
DB_PATH = "content.db"
ADMIN_TOKEN = os.environ.get("ADMIN_TOKEN", "dev-token")  # change this in Secrets

# ---------- DB ----------
def db():
    conn = sqlite3.connect(DB_PATH)
    conn.row_factory = sqlite3.Row
    return conn

def init_db():
    conn = db()
    conn.execute("""
      CREATE TABLE IF NOT EXISTS topics (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        keyword TEXT NOT NULL,
        intent TEXT NOT NULL,
        cluster TEXT,
        status TEXT NOT NULL DEFAULT 'queued', -- queued|drafted|published
        created_at TEXT NOT NULL
      )
    """)
    conn.execute("""
      CREATE TABLE IF NOT EXISTS drafts (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        topic_id INTEGER NOT NULL,
        title TEXT NOT NULL,
        slug TEXT NOT NULL,
        meta_description TEXT NOT NULL,
        content_html TEXT NOT NULL,
        faq_json TEXT,
        created_at TEXT NOT NULL,
        FOREIGN KEY(topic_id) REFERENCES topics(id)
      )
    """)
    conn.commit()
    conn.close()

init_db()

# ---------- AUTH ----------
def require_admin(req):
    token = req.args.get("token") or req.headers.get("X-Admin-Token")
    if token != ADMIN_TOKEN:
        abort(403)

# ---------- ROUTES ----------
@app.route("/")
def home():
    return "DigitalSignReviews Content Machine is running. Go to /admin?token=..."

@app.route("/topics/add", methods=["POST"])
def add_topic():
    require_admin(request)
    keyword = request.form.get("keyword", "").strip()
    intent = request.form.get("intent", "informational").strip()
    cluster = request.form.get("cluster", "").strip() or None
    if not keyword:
        abort(400, "keyword required")
    conn = db()
    conn.execute(
        "INSERT INTO topics(keyword,intent,cluster,status,created_at) VALUES (?,?,?,?,?)",
        (keyword, intent, cluster, "queued", datetime.utcnow().isoformat())
    )
    conn.commit()
    conn.close()
    return redirect(url_for("admin", token=ADMIN_TOKEN))

@app.route("/topics/generate/<int:topic_id>", methods=["POST"])
def generate(topic_id):
    require_admin(request)
    conn = db()
    topic = conn.execute("SELECT * FROM topics WHERE id=?", (topic_id,)).fetchone()
    if not topic:
        abort(404)
    draft = generate_draft(topic["keyword"], topic["intent"], topic["cluster"])

    conn.execute("""
      INSERT INTO drafts(topic_id,title,slug,meta_description,content_html,faq_json,created_at)
      VALUES (?,?,?,?,?,?,?)
    """, (topic_id, draft["title"], draft["slug"], draft["meta_description"],
          draft["content_html"], draft["faq_json"], datetime.utcnow().isoformat()))
    conn.execute("UPDATE topics SET status='drafted' WHERE id=?", (topic_id,))
    conn.commit()
    conn.close()
    return redirect(url_for("admin", token=ADMIN_TOKEN))

@app.route("/drafts/<int:draft_id>/publish", methods=["POST"])
def publish(draft_id):
    require_admin(request)
    # TODO: wire this to WordPress REST, Webflow, or whatever your CMS is.
    # For now we'll just mark as published.
    conn = db()
    draft = conn.execute("SELECT * FROM drafts WHERE id=?", (draft_id,)).fetchone()
    if not draft:
        abort(404)
    conn.execute("UPDATE topics SET status='published' WHERE id=?", (draft["topic_id"],))
    conn.commit()
    conn.close()
    return redirect(url_for("admin", token=ADMIN_TOKEN))

@app.route("/admin")
def admin():
    require_admin(request)
    conn = db()
    topics = conn.execute("SELECT * FROM topics ORDER BY id DESC LIMIT 50").fetchall()
    drafts = conn.execute("SELECT * FROM drafts ORDER BY id DESC LIMIT 20").fetchall()
    conn.close()
    return render_template_string(ADMIN_TEMPLATE, topics=topics, drafts=drafts, token=ADMIN_TOKEN)

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=3000, debug=True)
