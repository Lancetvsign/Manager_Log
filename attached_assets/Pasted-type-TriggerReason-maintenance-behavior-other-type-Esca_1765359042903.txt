type TriggerReason = "maintenance" | "behavior" | "other";

type EscalationRule = {
  id: number;
  name: string;
  triggerType: "reason" | "metric" | "flag";
  triggerReason?: TriggerReason;
  metricKey?: string;
  metricThreshold?: number | null;
  flagKey?: string;
  autoSend: boolean;
  enabled: boolean;
  presetKey?: string | null;
};

type AutomationPresetKey =
  | "maintenance_alert"
  | "behavior_alert"
  | "sales_drop_warning"
  | "gaming_outage";

type AutomationPreset = {
  key: AutomationPresetKey;
  title: string;
  description: string;
  defaultAutoSend: boolean;
  buildRule: (clubId: string, extra?: { threshold?: number }) => Omit<EscalationRule, "id">;
};
