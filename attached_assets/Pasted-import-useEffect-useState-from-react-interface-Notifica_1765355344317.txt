import { useEffect, useState } from "react";

interface NotificationList {
  id: number;
  clubId: string;
  name: string;
  description?: string | null;
  isDefault: boolean;
  recipientCount: number;
}

interface NotificationRecipient {
  id: number;
  listId: number;
  name?: string | null;
  email: string;
}

export function NotificationSettingsPage({ clubId }: { clubId: string }) {
  const [lists, setLists] = useState<NotificationList[]>([]);
  const [selectedListId, setSelectedListId] = useState<number | null>(null);
  const [recipients, setRecipients] = useState<NotificationRecipient[]>([]);
  const [newListName, setNewListName] = useState("");
  const [newRecipientName, setNewRecipientName] = useState("");
  const [newRecipientEmail, setNewRecipientEmail] = useState("");

  async function loadLists() {
    const res = await fetch(`/api/notifications/lists?clubId=${clubId}`);
    const data = await res.json();
    setLists(data);
    if (!selectedListId && data.length) {
      setSelectedListId(data[0].id);
    }
  }

  async function loadRecipients(listId: number) {
    const res = await fetch(`/api/notifications/lists/${listId}/recipients`);
    const data = await res.json();
    setRecipients(data);
  }

  useEffect(() => {
    loadLists();
  }, [clubId]);

  useEffect(() => {
    if (selectedListId) loadRecipients(selectedListId);
    else setRecipients([]);
  }, [selectedListId]);

  async function handleAddList() {
    if (!newListName.trim()) return;
    await fetch("/api/notifications/lists", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ clubId, name: newListName }),
    });
    setNewListName("");
    await loadLists();
  }

  async function handleAddRecipient() {
    if (!selectedListId || !newRecipientEmail.trim()) return;
    await fetch(`/api/notifications/lists/${selectedListId}/recipients`, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        name: newRecipientName || undefined,
        email: newRecipientEmail,
      }),
    });
    setNewRecipientName("");
    setNewRecipientEmail("");
    await loadRecipients(selectedListId);
  }

  return (
    <div className="grid grid-cols-[1.2fr_2fr] gap-6">
      {/* Lists column */}
      <section>
        <h1 className="text-lg font-semibold mb-2">Notification Lists</h1>
        <ul className="border rounded divide-y">
          {lists.map((list) => (
            <li
              key={list.id}
              className={
                "px-3 py-2 cursor-pointer flex justify-between items-center" +
                (list.id === selectedListId ? " bg-slate-100" : "")
              }
              onClick={() => setSelectedListId(list.id)}
            >
              <div>
                <div className="font-medium">{list.name}</div>
                <div className="text-xs text-muted-foreground">
                  {list.description || "No description"} ·{" "}
                  {list.recipientCount} recipients
                  {list.isDefault && " · Default"}
                </div>
              </div>
            </li>
          ))}
          {lists.length === 0 && (
            <li className="px-3 py-2 text-sm text-muted-foreground">
              No lists yet. Create your first list below.
            </li>
          )}
        </ul>

        <div className="mt-3 flex gap-2">
          <input
            className="flex-1 border rounded px-2 py-1 text-sm"
            placeholder="New list name (e.g. Core Team)"
            value={newListName}
            onChange={(e) => setNewListName(e.target.value)}
          />
          <button
            className="px-3 py-1 text-sm rounded bg-emerald-600 text-white"
            onClick={handleAddList}
          >
            Add
          </button>
        </div>
      </section>

      {/* Recipients column */}
      <section>
        <h2 className="text-lg font-semibold mb-2">Recipients</h2>
        {selectedListId ? (
          <>
            <table className="w-full text-sm border rounded overflow-hidden">
              <thead className="bg-slate-50 text-left">
                <tr>
                  <th className="px-3 py-2">Name</th>
                  <th className="px-3 py-2">Email</th>
                </tr>
              </thead>
              <tbody>
                {recipients.map((r) => (
                  <tr key={r.id} className="border-t">
                    <td className="px-3 py-1">{r.name || "—"}</td>
                    <td className="px-3 py-1">{r.email}</td>
                  </tr>
                ))}
                {recipients.length === 0 && (
                  <tr>
                    <td
                      colSpan={2}
                      className="px-3 py-2 text-muted-foreground"
                    >
                      No recipients yet. Add one below.
                    </td>
                  </tr>
                )}
              </tbody>
            </table>

            <div className="mt-3 flex gap-2">
              <input
                className="flex-1 border rounded px-2 py-1 text-sm"
                placeholder="Name (optional)"
                value={newRecipientName}
                onChange={(e) => setNewRecipientName(e.target.value)}
              />
              <input
                className="flex-1 border rounded px-2 py-1 text-sm"
                placeholder="Email"
                value={newRecipientEmail}
                onChange={(e) => setNewRecipientEmail(e.target.value)}
              />
              <button
                className="px-3 py-1 text-sm rounded bg-slate-900 text-white"
                onClick={handleAddRecipient}
              >
                Add
              </button>
            </div>
          </>
        ) : (
          <p className="text-sm text-muted-foreground">
            Select a list on the left to manage its recipients.
          </p>
        )}
      </section>
    </div>
  );
}
