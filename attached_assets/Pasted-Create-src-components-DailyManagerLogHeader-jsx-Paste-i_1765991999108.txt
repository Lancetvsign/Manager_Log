import React, { useMemo } from "react";

function formatLongDate(dateStr) {
  if (!dateStr) return "";
  const d = new Date(dateStr + "T00:00:00");
  return d.toLocaleDateString(undefined, { year: "numeric", month: "long", day: "numeric" });
}

function formatWeekday(dateStr) {
  if (!dateStr) return "";
  const d = new Date(dateStr + "T00:00:00");
  return d.toLocaleDateString(undefined, { weekday: "long" });
}

export default function DailyManagerLogHeader({
  date, setDate,
  amManager, setAmManager,
  pmManager, setPmManager,
  kitchenManager, setKitchenManager,
  dailyNumber,
  weeklyNumber,
  weatherTodayText = "—",
  weatherLastYearText = "—",
}) {
  const longDate = useMemo(() => formatLongDate(date), [date]);
  const weekday = useMemo(() => formatWeekday(date), [date]);

  return (
    <div className="rounded-xl border border-white/10 bg-white/5 p-5">
      <div className="flex items-start justify-between gap-4">
        <div>
          <h2 className="text-lg font-semibold text-white">Daily Manager Log Entry</h2>

          {/* Date (editable) + Day (read-only text) */}
          <div className="mt-4 flex flex-col gap-2">
            <label className="text-xs uppercase tracking-wide text-white/60">
              Date
            </label>

            <div className="flex flex-wrap items-center gap-3">
              <input
                type="date"
                value={date || ""}
                onChange={(e) => setDate?.(e.target.value)}
                className="h-9 w-[190px] rounded-lg border border-white/10 bg-black/20 px-3 text-sm text-white outline-none focus:border-white/25"
              />

              <div className="flex flex-col leading-tight">
                <div className="text-sm text-white/90">{longDate || "Select a date"}</div>
                <div className="text-xs text-white/60">{weekday || ""}</div>
              </div>
            </div>
          </div>
        </div>

        {/* Metadata + Weather chips (secondary) */}
        <div className="flex flex-col items-end gap-2">
          <div className="text-xs text-white/60">
            <span className="mr-3">Daily #: <span className="text-white/85">{dailyNumber ?? "—"}</span></span>
            <span>Weekly #: <span className="text-white/85">{weeklyNumber ?? "—"}</span></span>
          </div>

          <div className="flex flex-wrap items-center gap-2">
            <span className="rounded-full border border-white/10 bg-white/5 px-3 py-1 text-xs text-white/80">
              ☀ Today: {weatherTodayText}
            </span>
            <span className="rounded-full border border-white/10 bg-white/5 px-3 py-1 text-xs text-white/80">
              ❄ Last Year: {weatherLastYearText}
            </span>
          </div>
        </div>
      </div>

      {/* Managers grid */}
      <div className="mt-6">
        <div className="text-xs uppercase tracking-wide text-white/60">
          Managers on Duty
        </div>

        <div className="mt-3 grid grid-cols-1 gap-3 md:grid-cols-2">
          <ManagerRow label="AM" value={amManager} onChange={setAmManager} />
          <ManagerRow label="PM" value={pmManager} onChange={setPmManager} />
          <ManagerRow label="Kitchen" value={kitchenManager} onChange={setKitchenManager} />
        </div>
      </div>
    </div>
  );
}

function ManagerRow({ label, value, onChange }) {
  return (
    <div className="flex items-center gap-3 rounded-lg border border-white/10 bg-black/10 p-3">
      <div className="w-16 text-xs font-semibold uppercase tracking-wide text-white/70">
        {label}
      </div>
      <input
        value={value || ""}
        onChange={(e) => onChange?.(e.target.value)}
        placeholder="Name"
        className="h-9 w-full rounded-lg border border-white/10 bg-black/20 px-3 text-sm text-white outline-none focus:border-white/25"
      />
    </div>
  );
}

3) Make the “Day” stop being an input

Delete your old “Day” input entirely. This component computes it from the Date and shows it as read-only text (that’s part of why it feels more ergonomic).

4) Two tiny polish tweaks (optional but worth it)

If you want it a bit “slicker” on dark UI:

Change the outer background from bg-white/5 → bg-white/[0.04]

Change borders from border-white/10 → border-white/[0.08]

Small difference, but it reduces that “boxy” feel.
