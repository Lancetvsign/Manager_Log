Perfect — here’s a drop-in refactor for client/src/components/charts/SalesChart.tsx that:

merges current + comparison by shared label

renders current (solid) + comparison (dashed for line/area, lighter for bar)

fixes hasCompare by checking for compare_* presence

supports toggling Total / Food / Bar / Gaming and applies to both current + compare

Assumption: your series points have { label, total, food, bar, gaming }. If your keys differ, change the SalesPoint type + mapping in mergeSalesSeries().

// client/src/components/charts/SalesChart.tsx
import React, { useMemo } from "react";
import {
  ResponsiveContainer,
  ComposedChart,
  Line,
  Area,
  Bar,
  XAxis,
  YAxis,
  CartesianGrid,
  Tooltip,
  Legend,
} from "recharts";

type SalesPoint = {
  label: string;
  total: number;
  food: number;
  bar: number;
  gaming: number;
};

type MergedSalesPoint = {
  label: string;

  current_total: number;
  current_food: number;
  current_bar: number;
  current_gaming: number;

  // only present when compare is enabled
  compare_total?: number;
  compare_food?: number;
  compare_bar?: number;
  compare_gaming?: number;
};

type SeriesKey = "total" | "food" | "bar" | "gaming";
type ChartMode = "line" | "bar" | "area";

export type SalesChartProps = {
  data: SalesPoint[]; // current period series
  comparisonData?: SalesPoint[]; // comparison series (optional)
  mode?: ChartMode; // "line" | "bar" | "area"
  // which series should be shown
  visible?: Partial<Record<SeriesKey, boolean>>;
  height?: number;
};

function mergeSalesSeries(current: SalesPoint[], compare?: SalesPoint[]): MergedSalesPoint[] {
  const map = new Map<string, MergedSalesPoint>();

  for (const p of current) {
    map.set(p.label, {
      label: p.label,
      current_total: p.total ?? 0,
      current_food: p.food ?? 0,
      current_bar: p.bar ?? 0,
      current_gaming: p.gaming ?? 0,
    });
  }

  if (compare) {
    for (const p of compare) {
      const existing =
        map.get(p.label) ??
        ({
          label: p.label,
          current_total: 0,
          current_food: 0,
          current_bar: 0,
          current_gaming: 0,
        } as MergedSalesPoint);

      map.set(p.label, {
        ...existing,
        compare_total: p.total ?? 0,
        compare_food: p.food ?? 0,
        compare_bar: p.bar ?? 0,
        compare_gaming: p.gaming ?? 0,
      });
    }
  }

  // preserve current ordering
  return current.map((p) => map.get(p.label)!).filter(Boolean);
}

export default function SalesChart({
  data,
  comparisonData,
  mode = "line",
  visible,
  height = 320,
}: SalesChartProps) {
  const show: Record<SeriesKey, boolean> = {
    total: visible?.total ?? true,
    food: visible?.food ?? false,
    bar: visible?.bar ?? false,
    gaming: visible?.gaming ?? false,
  };

  const mergedData = useMemo(() => mergeSalesSeries(data ?? [], comparisonData), [data, comparisonData]);

  const hasCompare = useMemo(() => {
    // if any compare_* key exists (not undefined), we consider compare enabled
    return mergedData.some(
      (p) =>
        p.compare_total !== undefined ||
        p.compare_food !== undefined ||
        p.compare_bar !== undefined ||
        p.compare_gaming !== undefined
    );
  }, [mergedData]);

  // ---- helpers to render series consistently ----
  const renderLineOrArea = (key: SeriesKey) => {
    const currentKey = `current_${key}` as const;
    const compareKey = `compare_${key}` as const;

    if (!show[key]) return null;

    if (mode === "area") {
      return (
        <React.Fragment key={key}>
          <Area type="monotone" dataKey={currentKey} dot={false} />
          {hasCompare && (
            <Line
              type="monotone"
              dataKey={compareKey}
              dot={false}
              strokeDasharray="6 6"
              opacity={0.7}
            />
          )}
        </React.Fragment>
      );
    }

    // line
    return (
      <React.Fragment key={key}>
        <Line type="monotone" dataKey={currentKey} dot={false} />
        {hasCompare && (
          <Line
            type="monotone"
            dataKey={compareKey}
            dot={false}
            strokeDasharray="6 6"
            opacity={0.7}
          />
        )}
      </React.Fragment>
    );
  };

  const renderBars = (key: SeriesKey) => {
    const currentKey = `current_${key}` as const;
    const compareKey = `compare_${key}` as const;

    if (!show[key]) return null;

    // Bars can’t be dashed well; use lighter opacity for comparison
    return (
      <React.Fragment key={key}>
        <Bar dataKey={currentKey} />
        {hasCompare && <Bar dataKey={compareKey} opacity={0.35} />}
      </React.Fragment>
    );
  };

  return (
    <div className="w-full">
      <div style={{ width: "100%", height }}>
        <ResponsiveContainer>
          <ComposedChart data={mergedData} margin={{ top: 10, right: 16, left: 0, bottom: 0 }}>
            <CartesianGrid strokeDasharray="3 3" opacity={0.25} />
            <XAxis dataKey="label" />
            <YAxis />
            <Tooltip />
            <Legend />

            {/* Render series */}
            {mode === "bar" ? (
              <>
                {renderBars("total")}
                {renderBars("food")}
                {renderBars("bar")}
                {renderBars("gaming")}
              </>
            ) : (
              <>
                {renderLineOrArea("total")}
                {renderLineOrArea("food")}
                {renderLineOrArea("bar")}
                {renderLineOrArea("gaming")}
              </>
            )}
          </ComposedChart>
        </ResponsiveContainer>
      </div>

      {/* Optional: tiny hint text */}
      {hasCompare && (
        <div className="mt-2 text-xs text-slate-400">
          Solid = current period • Dashed/lighter = comparison period
        </div>
      )}
    </div>
  );
}

What you may need to adjust (quick)

If your data keys aren’t total/food/bar/gaming, change SalesPoint and mergeSalesSeries.

If your chart uses only one selected series instead of multiple, just flip visible to only true for the chosen one.

If you have “Total” already precomputed elsewhere, keep it — or compute it before passing into the chart.
