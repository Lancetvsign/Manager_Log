import { useEffect, useMemo, useState } from "react";

type NotificationList = {
  id: string;
  name: string;
  recipientCount: number;
};

type Props = {
  open: boolean;
  onOpenChange: (v: boolean) => void;
  clubId: string;
  logId: string | null;
  logDateISO: string;
  managerName?: string;
};

export default function EmailLogModal({
  open,
  onOpenChange,
  clubId,
  logId,
  logDateISO,
  managerName,
}: Props) {
  const [lists, setLists] = useState<NotificationList[]>([]);
  const [loadingLists, setLoadingLists] = useState(false);

  const [selectedListIds, setSelectedListIds] = useState<string[]>([]);
  const [reason, setReason] = useState<"maintenance" | "operations" | "other">("maintenance");

  const defaultSubject = useMemo(() => {
    const niceDate = new Date(logDateISO).toLocaleDateString(undefined, {
      year: "numeric",
      month: "long",
      day: "numeric",
    });
    const label =
      reason === "maintenance" ? "Urgent maintenance" :
      reason === "operations" ? "Operational issue" :
      "Urgent item";
    return `[Club] ${niceDate} - ${label}`;
  }, [logDateISO, reason]);

  const [subject, setSubject] = useState("");
  const [message, setMessage] = useState("");

  // settings section state
  const [settingsTab, setSettingsTab] = useState<"contacts" | "rules">("contacts");

  const [sending, setSending] = useState(false);
  const [status, setStatus] = useState<{ type: "ok" | "err"; text: string } | null>(null);

  useEffect(() => {
    if (!open) return;

    setStatus(null);
    setSubject(defaultSubject);
    setMessage(
`Date: ${logDateISO}
Manager: ${managerName || "Not specified"}

Summary:
- (type here)

Details:
- (optional)

A link to this log entry will be included automatically.`
    );

    (async () => {
      setLoadingLists(true);
      try {
        const res = await fetch(`/api/notification-lists?clubId=${encodeURIComponent(clubId)}`);
        const data = await res.json();
        setLists(Array.isArray(data) ? data : []);
        // default select "Core Team" if exists
        const core = (Array.isArray(data) ? data : []).find((x: any) =>
          (x.name || "").toLowerCase().includes("core")
        );
        setSelectedListIds(core?.id ? [core.id] : []);
      } catch {
        setLists([]);
      } finally {
        setLoadingLists(false);
      }
    })();
  }, [open, clubId, defaultSubject, logDateISO, managerName]);

  if (!open) return null;

  const canEmail = !!logId;

  const toggleList = (id: string) => {
    setSelectedListIds((prev) => (prev.includes(id) ? prev.filter((x) => x !== id) : [...prev, id]));
  };

  async function onSend(e: React.MouseEvent<HTMLButtonElement>) {
    e.preventDefault();
    e.stopPropagation();

    if (!canEmail || !logId) {
      setStatus({ type: "err", text: "Save the log entry first to enable emailing." });
      return;
    }
    if (selectedListIds.length === 0) {
      setStatus({ type: "err", text: "Pick at least one notification list." });
      return;
    }

    setSending(true);
    setStatus(null);
    try {
      const res = await fetch(`/api/notify-core-team`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          clubId,
          logId,
          reason,
          subject,
          message,
          listIds: selectedListIds,
        }),
      });

      if (!res.ok) throw new Error(await res.text());
      setStatus({ type: "ok", text: "Sent successfully." });
    } catch (err: any) {
      setStatus({ type: "err", text: err?.message || "Failed to send email." });
    } finally {
      setSending(false);
    }
  }

  return (
    <div
      className="fixed inset-0 z-50 flex items-center justify-center bg-black/50 p-4"
      onMouseDown={() => onOpenChange(false)}
    >
      <div
        className="w-full max-w-3xl rounded-xl bg-background p-4 shadow-xl"
        onMouseDown={(e) => e.stopPropagation()}
      >
        <div className="flex items-center justify-between">
          <div className="text-lg font-semibold">📧 Email this log</div>
          <button
            type="button"
            className="rounded-md border px-3 py-1 text-sm"
            onClick={() => onOpenChange(false)}
          >
            Close
          </button>
        </div>

        {!canEmail && (
          <div className="mt-3 rounded-md bg-amber-50 px-3 py-2 text-sm text-amber-900">
            Save this log to enable emailing.
          </div>
        )}

        <div className="mt-4 grid gap-4">
          {/* Reason */}
          <div className="flex flex-wrap items-center gap-4">
            <div className="text-sm font-medium">Reason</div>
            <label className="flex items-center gap-2 text-sm">
              <input type="radio" checked={reason === "maintenance"} onChange={() => setReason("maintenance")} />
              Maintenance
            </label>
            <label className="flex items-center gap-2 text-sm">
              <input type="radio" checked={reason === "operations"} onChange={() => setReason("operations")} />
              Operations
            </label>
            <label className="flex items-center gap-2 text-sm">
              <input type="radio" checked={reason === "other"} onChange={() => setReason("other")} />
              Other
            </label>
          </div>

          {/* Send to */}
          <div>
            <div className="mb-2 text-sm font-medium">Send to</div>
            <div className="rounded-md border p-3">
              {loadingLists ? (
                <div className="text-sm text-muted-foreground">Loading lists…</div>
              ) : lists.length === 0 ? (
                <div className="text-sm text-muted-foreground">No lists yet. Add lists in Settings → Contacts.</div>
              ) : (
                <div className="grid gap-2">
                  {lists.map((l) => (
                    <label key={l.id} className="flex items-center justify-between gap-3 text-sm">
                      <div className="flex items-center gap-2">
                        <input
                          type="checkbox"
                          checked={selectedListIds.includes(l.id)}
                          onChange={() => toggleList(l.id)}
                        />
                        <span>{l.name}</span>
                      </div>
                      <span className="text-xs text-muted-foreground">{l.recipientCount} recipients</span>
                    </label>
                  ))}
                </div>
              )}
            </div>
          </div>

          {/* Subject */}
          <div>
            <div className="mb-2 text-sm font-medium">Subject</div>
            <input
              className="w-full rounded-md border bg-transparent px-3 py-2 text-sm"
              value={subject}
              onChange={(e) => setSubject(e.target.value)}
            />
          </div>

          {/* Message */}
          <div>
            <div className="mb-2 text-sm font-medium">Message</div>
            <textarea
              className="min-h-[160px] w-full rounded-md border bg-transparent px-3 py-2 text-sm"
              value={message}
              onChange={(e) => setMessage(e.target.value)}
            />
            <div className="mt-1 text-xs text-muted-foreground">
              A link to this log entry will be included automatically.
            </div>
          </div>

          {/* SETTINGS: contacts + rules inside here */}
          <details className="rounded-md border p-3">
            <summary className="cursor-pointer text-sm font-medium">Settings</summary>

            <div className="mt-3">
              <div className="flex gap-2">
                <button
                  type="button"
                  className={`rounded-md border px-3 py-1 text-sm ${settingsTab === "contacts" ? "bg-accent" : ""}`}
                  onClick={() => setSettingsTab("contacts")}
                >
                  Manage Contacts
                </button>
                <button
                  type="button"
                  className={`rounded-md border px-3 py-1 text-sm ${settingsTab === "rules" ? "bg-accent" : ""}`}
                  onClick={() => setSettingsTab("rules")}
                >
                  Rules
                </button>
              </div>

              <div className="mt-3 rounded-md border p-3">
                {settingsTab === "contacts" ? (
                  <div className="text-sm text-muted-foreground">
                    {/* Replace this block by rendering your existing Notification Settings component */}
                    Drop your “Notification Lists” UI here (create list, add/remove recipients).
                    <br />
                    Recommended: render the same component you used in the old Settings tab.
                  </div>
                ) : (
                  <div className="text-sm text-muted-foreground">
                    {/* Replace this block by rendering your existing Rules/Smart Automations component */}
                    Drop your “Smart Automations / Escalation Rules” UI here.
                  </div>
                )}
              </div>
            </div>
          </details>

          {status && (
            <div
              className={`rounded-md px-3 py-2 text-sm ${
                status.type === "ok" ? "bg-emerald-50 text-emerald-900" : "bg-red-50 text-red-900"
              }`}
            >
              {status.text}
            </div>
          )}

          <div className="flex items-center justify-end gap-2">
            <button type="button" className="rounded-md border px-4 py-2 text-sm" onClick={() => onOpenChange(false)}>
              Cancel
            </button>
            <button
              type="button"
              className="rounded-md bg-red-600 px-4 py-2 text-sm text-white disabled:opacity-50"
              disabled={!canEmail || sending}
              onClick={onSend}
            >
              {sending ? "Sending…" : "Send"}
            </button>
          </div>
        </div>
      </div>
    </div>
  );
}
