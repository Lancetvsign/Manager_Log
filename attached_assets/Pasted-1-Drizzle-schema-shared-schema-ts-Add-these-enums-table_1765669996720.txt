// server/routes.ts
import type { Express } from "express";
import fs from "fs";
import path from "path";
import multer from "multer";
import { db } from "./db";
import { managerLogs, logAttachments } from "@shared/schema";
import { and, eq, desc } from "drizzle-orm";

const uploadsDir = path.join(process.cwd(), "uploads");
if (!fs.existsSync(uploadsDir)) fs.mkdirSync(uploadsDir, { recursive: true });

// Disk storage (dev-friendly)
const storage = multer.diskStorage({
  destination: (_req, _file, cb) => cb(null, uploadsDir),
  filename: (_req, file, cb) => {
    const safe = file.originalname.replace(/[^a-zA-Z0-9._-]/g, "_");
    cb(null, `${Date.now()}_${Math.random().toString(16).slice(2)}_${safe}`);
  },
});

const upload = multer({
  storage,
  limits: { fileSize: 10 * 1024 * 1024 }, // 10MB
});

function normalizeDate(input: string) {
  // expects "YYYY-MM-DD" coming from client; if you pass Date objects, normalize there.
  return input?.slice(0, 10);
}

async function getOrCreateDraftLog(clubId: number, date: string) {
  const d = normalizeDate(date);

  const existing = await db
    .select({ id: managerLogs.id, status: managerLogs.status })
    .from(managerLogs)
    .where(and(eq(managerLogs.clubId, clubId), eq(managerLogs.date, d)))
    .limit(1);

  if (existing.length) return existing[0].id;

  const inserted = await db
    .insert(managerLogs)
    .values({
      clubId,
      date: d,
      status: "draft",
      // optionally prefill managerName etc
    })
    .returning({ id: managerLogs.id });

  return inserted[0].id;
}

export function registerRoutes(app: Express) {
  /**
   * 1) Create or return a draft log ID so uploads can attach immediately
   * POST /api/logs/draft  { clubId, date }
   */
  app.post("/api/logs/draft", async (req, res) => {
    try {
      const { clubId, date } = req.body || {};
      if (!clubId || !date) return res.status(400).json({ error: "clubId and date required" });

      const logId = await getOrCreateDraftLog(Number(clubId), String(date));
      res.json({ logId });
    } catch (err: any) {
      console.error(err);
      res.status(500).json({ error: "Failed to create draft log" });
    }
  });

  /**
   * Serve uploaded files (dev). In production, replace with S3/Supabase public URLs.
   */
  app.use("/uploads", (req, res, next) => {
    // basic static handler
    const filePath = path.join(uploadsDir, req.path);
    if (!filePath.startsWith(uploadsDir)) return res.status(400).end();
    if (!fs.existsSync(filePath)) return res.status(404).end();
    res.sendFile(filePath);
  });

  /**
   * 2) Upload an attachment to a log + section
   * POST /api/logs/:logId/attachments  (multipart/form-data)
   * fields: section, file
   */
  app.post(
    "/api/logs/:logId/attachments",
    upload.single("file"),
    async (req, res) => {
      try {
        const logId = Number(req.params.logId);
        const { section, clubId } = req.body || {};
        const file = req.file;

        if (!logId) return res.status(400).json({ error: "logId required" });
        if (!clubId) return res.status(400).json({ error: "clubId required" });
        if (!section) return res.status(400).json({ error: "section required" });
        if (!file) return res.status(400).json({ error: "file required" });

        // Security check: log belongs to club
        const logCheck = await db
          .select({ id: managerLogs.id })
          .from(managerLogs)
          .where(and(eq(managerLogs.id, logId), eq(managerLogs.clubId, Number(clubId))))
          .limit(1);

        if (!logCheck.length) return res.status(403).json({ error: "Forbidden" });

        const storagePath = `/uploads/${file.filename}`;

        const inserted = await db
          .insert(logAttachments)
          .values({
            clubId: Number(clubId),
            logId,
            section,
            storagePath,
            originalName: file.originalname,
            mimeType: file.mimetype,
            size: file.size,
          })
          .returning();

        res.json({ attachment: inserted[0] });
      } catch (err: any) {
        console.error(err);
        res.status(500).json({ error: "Upload failed" });
      }
    }
  );

  /**
   * 3) List attachments by log + section
   * GET /api/logs/:logId/attachments?clubId=1&section=operational
   */
  app.get("/api/logs/:logId/attachments", async (req, res) => {
    try {
      const logId = Number(req.params.logId);
      const clubId = Number(req.query.clubId);
      const section = String(req.query.section || "");

      if (!logId || !clubId || !section) {
        return res.status(400).json({ error: "logId, clubId, section required" });
      }

      const items = await db
        .select()
        .from(logAttachments)
        .where(
          and(
            eq(logAttachments.logId, logId),
            eq(logAttachments.clubId, clubId),
            eq(logAttachments.section, section as any)
          )
        )
        .orderBy(desc(logAttachments.createdAt));

      res.json({ attachments: items });
    } catch (err: any) {
      console.error(err);
      res.status(500).json({ error: "Failed to load attachments" });
    }
  });

  /**
   * 4) Delete an attachment (and the file)
   * DELETE /api/attachments/:id?clubId=1
   */
  app.delete("/api/attachments/:id", async (req, res) => {
    try {
      const id = Number(req.params.id);
      const clubId = Number(req.query.clubId);

      if (!id || !clubId) return res.status(400).json({ error: "id and clubId required" });

      const found = await db
        .select()
        .from(logAttachments)
        .where(and(eq(logAttachments.id, id), eq(logAttachments.clubId, clubId)))
        .limit(1);

      if (!found.length) return res.status(404).json({ error: "Not found" });

      const attachment = found[0];
      // delete row first
      await db.delete(logAttachments).where(eq(logAttachments.id, id));

      // delete file
      const diskPath = path.join(process.cwd(), attachment.storagePath.replace("/uploads/", "uploads/"));
      if (diskPath.startsWith(uploadsDir) && fs.existsSync(diskPath)) fs.unlinkSync(diskPath);

      res.json({ ok: true });
    } catch (err: any) {
      console.error(err);
      res.status(500).json({ error: "Delete failed" });
    }
  });
}

B) Add a reusable uploader component

Create client/src/components/LogSectionUploader.tsx

import { useEffect, useState } from "react";

type Attachment = {
  id: number;
  storagePath: string;
  originalName: string;
  mimeType: string;
  createdAt: string;
};

export function LogSectionUploader({
  clubId,
  logId,
  section,
  title = "Photos",
}: {
  clubId: number;
  logId: number | null;
  section: "operational" | "maintenance";
  title?: string;
}) {
  const [attachments, setAttachments] = useState<Attachment[]>([]);
  const [loading, setLoading] = useState(false);

  async function load() {
    if (!logId) return;
    setLoading(true);
    const res = await fetch(
      `/api/logs/${logId}/attachments?clubId=${clubId}&section=${section}`
    );
    const data = await res.json();
    setAttachments(data.attachments || []);
    setLoading(false);
  }

  useEffect(() => {
    load();
  }, [logId, section]);

  async function onPickFiles(files: FileList | null) {
    if (!files || !logId) return;

    // upload sequentially (simple + reliable)
    for (const file of Array.from(files)) {
      const fd = new FormData();
      fd.append("clubId", String(clubId));
      fd.append("section", section);
      fd.append("file", file);

      await fetch(`/api/logs/${logId}/attachments`, {
        method: "POST",
        body: fd,
      });
    }

    await load();
  }

  async function remove(id: number) {
    await fetch(`/api/attachments/${id}?clubId=${clubId}`, { method: "DELETE" });
    await load();
  }

  return (
    <div className="mt-3">
      <div className="d-flex align-items-center justify-content-between">
        <div className="fw-semibold">{title}</div>
        <label className="btn btn-sm btn-outline-secondary mb-0">
          Add Photo
          <input
            type="file"
            accept="image/*"
            multiple
            style={{ display: "none" }}
            onChange={(e) => onPickFiles(e.target.files)}
          />
        </label>
      </div>

      {loading ? (
        <div className="text-muted small mt-2">Loading…</div>
      ) : attachments.length === 0 ? (
        <div className="text-muted small mt-2">
          No photos yet. Upload anytime — no need to save first.
        </div>
      ) : (
        <div className="d-flex flex-wrap gap-2 mt-2">
          {attachments.map((a) => {
            const isImg = a.mimeType?.startsWith("image/");
            return (
              <div
                key={a.id}
                className="border rounded p-2"
                style={{ width: 160 }}
              >
                {isImg ? (
                  <img
                    src={a.storagePath}
                    alt={a.originalName}
                    style={{ width: "100%", height: 90, objectFit: "cover", borderRadius: 6 }}
                  />
                ) : (
                  <div className="small">{a.originalName}</div>
                )}

                <div className="d-flex justify-content-between align-items-center mt-2">
                  <div className="small text-truncate" style={{ maxWidth: 110 }}>
                    {a.originalName}
                  </div>
                  <button
                    type="button"
                    className="btn btn-sm btn-outline-danger"
                    onClick={() => remove(a.id)}
                  >
                    ✕
                  </button>
                </div>
              </div>
            );
          })}
        </div>
      )}
    </div>
  );
}
